type 'raw_test project_params = {
  1. cil : Cil.file;
    (*

    CIL representation for the whole C project

    *)
  2. cil_typing_info : c_types_info;
  3. typdecls : Sc_values.typdecls;
    (*

    type declarations, derived from cil

    *)
  4. func_repr : Sc_C.Types.func_repr;
    (*

    representation of the entrypoint function; guaranteed not to be `main`

    *)
  5. test_struct : Sc_values.Struct.typ;
    (*

    struct type that gathers all test inputs for the entrypoing

    *)
  6. test_repr : 'raw_test Sc_corpus.Types.repr;
    (*

    module for test inputs representation

    *)
  7. init_func : Sc_C.Types.func_repr option;
    (*

    optional test initializer

    *)
  8. oracle_func : Sc_C.Types.func_repr option;
    (*

    optional test oracle

    *)
  9. seek_oracle_failures : bool;
    (*

    whether analysis tools should seek tests that fail the oracle

    *)
}

type 'raw_test project = {
  1. config : project_config;
    (*

    given configuration

    *)
  2. params : 'raw_test project_params;
    (*

    computed parameters

    *)
  3. workspace : Sc_core.Types.workspace;
    (*

    (== config.project_workspace)

    *)
  4. codebase : [ `C ] Sc_sys.File.TYPES.file;
    (*

    Never defines or use symbol `main`

    *)
  5. types_header : [ `h ] Sc_sys.File.TYPES.file;
    (*

    C header file with declaration of types from the codebase

    *)
  6. func_header : [ `h ] Sc_sys.File.TYPES.file;
    (*

    C header file with declaration of entrypoint and globals

    *)
  7. labels : Sc_ltest.Types.labels;
    (*

    managed labels (could be in parameters)

    *)
  8. label_data : Sc_ltest.Types.label_data;
    (*

    managed label data

    *)
  9. store : Sc_store.t;
    (*

    store of coverage statuses

    *)
  10. decoder : 'raw_test Sc_corpus.Decoder.t;
    (*

    dynamic raw test decoder

    *)
  11. validator : 'raw_test Sc_corpus.Validator.t;
  12. corpus : 'raw_test Sc_corpus.corpus;
    (*

    corpus of inputs

    *)
  13. outdir : Sc_sys.File.TYPES.dir;
    (*

    output directory for generated C files

    *)
  14. covdir : Sc_sys.File.TYPES.dir;
    (*

    output directory for label-covering C files (subdirectory of outdir)

    *)
  15. rtedir : Sc_sys.File.TYPES.dir;
    (*

    output directory for RTE-triggering C files (subdirectory of outdir)

    *)
  16. faildir : Sc_sys.File.TYPES.dir option;
    (*

    output directory for C files that fail the oracle (subdirectory of outdir)

    *)
  17. stats : statistics;
    (*

    various tool statistics

    *)
  18. extra : project_extra;
    (*

    additional data used for project management

    *)
}

type testsuite =
  1. | Split_testsuite of {
    1. cov : individual_test_in_testsuite list;
    2. covdir : Sc_sys.File.TYPES.dir;
    3. rte : individual_test_in_testsuite list;
    4. rtedir : Sc_sys.File.TYPES.dir;
    5. fail : individual_test_in_testsuite list option;
    6. faildir : Sc_sys.File.TYPES.dir option;
    }
  2. | Combined_testsuite of {
    1. cov : combined_tests_in_testsuite;
    2. covdir : Sc_sys.File.TYPES.dir;
    3. rte : combined_tests_in_testsuite;
    4. rtedir : Sc_sys.File.TYPES.dir;
    5. fail : combined_tests_in_testsuite option;
    6. faildir : Sc_sys.File.TYPES.dir option;
    }

}

Errors & exceptions

type preproc_error =
  1. | Missing_entrypoint of string
  2. | Missing_cover_target of string
  3. | No_inputs
type setup_error =
  1. | Ltest of Sc_ltest.Types.error
  2. | Preproc of preproc_error NEL.t
exception SETUP_ERROR of setup_error
type labeling_error =
  1. | Failed_labeling of exn Sc_core.Types.generic_preprocessing_error
  2. | Syntax_errors of {
    1. cmd_error : Sc_C.Cmd.error;
    2. stderr_lines : string list;
    }
exception LABELING_ERROR of labeling_error
type elaboration_error =
  1. | Failed_elaboration of exn
  2. | Unknown_formals of {
    1. formals : Basics.Strings.t;
    2. func : Sc_C.Types.func_repr;
    }
  3. | Unknown_function of {
    1. fun_name : string;
    2. c_file : [ `C ] Sc_sys.File.TYPES.file option;
    }
  4. | Unsupported_formals of {
    1. formals : Basics.Strings.t;
    2. func : Sc_C.Types.func_repr;
    }
  5. | Unexpected_initialization_function_with_args of {
    1. func : Sc_C.Types.func_repr;
    }
exception ELABORATION_ERROR of elaboration_error
