type run_config = {
  1. config_input : Sc_config.Eztoml.TYPES.toml_file option;
    (*

    TOML configuration file, either given via command-line arguments, or found automatically based on the current directory

    *)
  2. workdir : string;
    (*

    name of directory where all the working files will be stored ("_sc" by default)

    *)
  3. force_preprocess : bool;
    (*

    force project elaboration, labeling and tool preprocessing stages

    *)
  4. enable_syntax_check : bool;
    (*

    whether to perform a syntax check prior to labeling

    *)
  5. test_timeout : float;
  6. tools : string list;
    (*

    tools to use

    *)
  7. strategy : orchestration_strategy;
    (*

    tool orchestration strategy

    *)
  8. max_validation_concurrency : int;
    (*

    maximum number of validations that may be performed in parallel

    *)
  9. verbose_validation : bool;
  10. display_outcomes : [ `No | `Yes | `Verbose ];
}
and orchestration_strategy =
  1. | AllParallel
    (*

    launch every tool in parallel

    *)
  2. | AllSequential
    (*

    launch tools sequentially

    *)
  3. | Optimized
    (*

    use a heuristic to launch tools

    *)
  4. | Custom of Sc_strategy.Types.t
    (*

    use-provided strategy

    *)

type project_config = {
  1. name : string;
    (*

    name for the project; may be "" (the default)

    *)
  2. input_files : string list;
    (*

    C files of the project

    *)
  3. entrypoint : string;
    (*

    function to analyze

    *)
  4. cover_criterion : string;
    (*

    Lannot name for criterion

    *)
  5. functions_to_cover : [ `All | `Only of string list | `Auto ];
  6. header_dirs : string list;
    (*

    dirs where C header files are located

    *)
  7. external_libs : string list;
  8. ignored_globals : string list;
}
