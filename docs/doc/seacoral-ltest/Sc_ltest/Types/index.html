type config = {
  1. input_files : [ `C ] Sc_sys.File.TYPES.file list;
    (*

    given C files

    *)
  2. entrypoint : string;
    (*

    single entrypoint function to be tested

    *)
  3. criterion : string;
    (*

    coverage criterion (see frama-c -lannot-list)

    *)
  4. header_dirs : Sc_sys.File.TYPES.dir list;
    (*

    C header directories

    *)
  5. external_libs : string list;
    (*

    external library names

    *)
  6. annotated_functions : [ `All | `Only of string list | `Auto ];
    (*

    specification of functions to consider for coverage

    *)
  7. ignored_globals : Basics.Strings.t;
  8. fixtures_files : [ `C ] Sc_sys.File.TYPES.file list;
    (*

    additional fixture files

    *)
  9. initialization_function : string;
  10. oracle_function : string;
  11. seek_oracle_failures : bool;
    (*

    whether analysis tools should seek tests that fail the oracle

    *)
}
