A type that enables tools to either retrieve an actual test inputs value (via raw), or to create a filesystem link with a file containing this value (via link).

and test_metadata = {
  1. serialnum : int;
    (*

    Ordinal (starting from 1) incremented with every new registration of a test into the corpus

    *)
  2. toolname : string;
    (*

    Name of the tool that first transmitted the test

    *)
  3. creatime : float;
    (*

    Registration time (using local timezone)

    *)
  4. crearun : int;
    (*

    Run during which the test was created

    *)
  5. id : Stdlib.Digest.t;
    (*

    Test digest (computed on internal representation)

    *)
  6. outcome : test_outcome;
    (*

    Purpose of the test (covering labels, RTE)

    *)
}
and test_outcome =
  1. | Covering_label of Basics.Ints.t
    (*

    The test does not raise an RTE and Covers labels

    *)
  2. | Triggering_RTE of sanitizer_error_summary
    (*

    The test triggers an RTE

    *)
  3. | Oracle_failure
    (*

    The test fails the oracle

    *)
and sanitizer_error_summary =
  1. | Heap_buffer_overflow of int64
  2. | Invalid_memory_address of int64
  3. | Arithmetic_error of int64
