Module type Struct.VAL

module Printer : sig ... end
type t
val blank : typ -> t

blank typ returns a new value of type typ with all bytes set to null (zero).

val of_string : typ -> string -> t

of_string typ str interprets the string of characters str as a C structured value of type typ. Raises Invalid_value_representation if str is not long-enough to hold such a value.

val to_string : t -> string

to_string typ v returns the C structured value v, of type typ, as a string of bytes encoded in a string.

val read : typ -> Stdlib.in_channel -> t

read typ ic acts like of_string, but reads bytes from the input channel ic.

val write : Stdlib.out_channel -> t -> unit

write oc v writes the binary representation of the C structured value v onto oc.

val typ : t -> typ

typ v returns the type of v.

val print : Stdlib.Format.formatter -> t -> unit

print fmt v prints the structured value v as a C-like literal using the formatter fmt; in the presence of pointers, this does not generate parseable C code.

val as_c_literal : ?lit_heap:(Printer.literal_heap as 'm) -> t -> string * 'm

as_c_literal ?lit_heap v constructs a parseable C literal representation for the value v, along with a possibly empty set of object literals representing cells of dynamic memory.

val assign_from_literal : Sc_values__.Cvalues.typdecls -> t -> Sc_values__.Cvalues.literal_binding -> unit

assign_from_literal v bindings performs a deep assignent of v w.r.t the bindings described using bindings.

val fold_pointers : deep:bool -> f:('a -> 'a) Sc_values__.Cvalues.TYPES.cptr_consumer -> t -> 'a -> 'a
val fields_as_c_literals : t -> Printer.literal_memory

fields_as_c_literals v returns a pair (fields, heap): fields is a list of pairs of strings (fname, fval), where fname is the name of a field in the type of v, and fval is a human-readable, C-parseable, representation of its value in v; in turn, heap is a possibly empty literal heap.

val fields_as_c_allocations : t -> Printer.initialization_block

fields_as_c_allocations v returns a list of instructions that initializes one variable f for each field f of v. Each heap object pointed to via a field that is (transitively) reachable from any f is dynamically allocated.
