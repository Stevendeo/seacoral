Module Config.TYPES

type options = {
  1. timeout : float;
  2. only_new_tests : bool;
  3. keep_error_cases : bool;
  4. search : string list;
  5. label_handling : label_handling;
  6. replay : replay;
  7. replay_max_concurrency : int;
  8. libc : string;
}
and replay =
  1. | Inhibit
    (*

    do not replay any test

    *)
  2. | Enable
    (*

    on-the-fly replay if relevant (w.r.t label handling policy)

    *)
  3. | Delayed
    (*

    delay any replay until after the main klee process terminates

    *)
  4. | Force
    (*

    force on-the-fly replay, whatever the label handling policy

    *)
and label_handling =
  1. | Ignore
    (*

    pc_label(e, id) amounts to no C code

    *)
  2. | Naive
    (*

    pc_label(e, id) amounts to if e set_cover(id)

    *)
  3. | Optimized of {
    1. benign_test_policy : benign_test_policy;
    }
and benign_test_policy =
  1. | Keep
    (*

    generate and keep all tests

    *)
  2. | Reject
    (*

    do not keep any test that does not reach a label

    *)
  3. | Avoid
    (*

    do not generate any test that neither reaches a label nor triggers a runtime error

    *)
