Access path

type access_path_operation =
  1. | Field_access of string
    (*

    access to a field given by its name

    *)
  2. | Index_access of int
    (*

    access to an array cell given by its index

    *)

A concrete access path is built out of simple access operations

type access_path_suffix = access_path_operation NEL.t

An access path suffix is a non-empty list of access path operations

type access_path =
  1. | Access_path_origin of string
  2. | Access_path_suffix of access_path_suffix
  3. | Access_path of {
    1. origin : string;
    2. suffix : access_path_suffix;
    }

A concrete access path combines an origin variable name and/or a suffix

Coverage labels

type cov_label_view = {
  1. cov_label_id : int;
    (*

    The unique ID of the proof objective

    *)
  2. cov_label_status : cov_label_status;
    (*

    The current status of the proof objective

    *)
  3. cov_label_orig_loc : location;
    (*

    The location of the proof objective in the original file

    *)
  4. cov_label_loc : location;
    (*

    The location in the labelized file

    *)
  5. cov_label_emitter : string option;
    (*

    The tool that emited the status

    *)
  6. cov_label_exec_time : float;
    (*

    Exec time of the proof

    *)
}
and location = {
  1. loc_file : string;
  2. loc_line : int;
}

A location is a line in a file.

and cov_label_status =
  1. | Unknown
  2. | Covered of string list
  3. | Uncoverable

Status of a coverage label / proof objective

type simple = [
  1. | `simple
]
type hyper = [
  1. | `hyper
]
type any = [
  1. | simple
  2. | hyper
  3. | `any
]
type _ cov_label =
  1. | S : cov_label_view -> [> simple ] cov_label
  2. | H : cov_label_view -> [> hyper ] cov_label
  3. | Any : [ simple | hyper ] cov_label -> [> `any ] cov_label
type pointer_refs = pointer_ref list

Representation for sets of pointer references

and pointer_ref =
  1. | Variable of {
    1. pointer_var : string;
    }
    (*

    Reference to a pointer variable

    *)
  2. | Struct_field of {
    1. struct_name : string;
    2. pointer_field_name : string;
    }
    (*

    Reference to a pointer field

    *)

Type of pointer references

type pointer_constraints = pointer_constraint list

Representation for pointer constraints

and pointer_constraint =
  1. | Distinct_variables of {
    1. pointer_var : string;
      (*

      Name of the constrained pointer variable

      *)
    2. size_var : string;
      (*

      Name of the constraining size variable

      *)
    }
    (*

    Constraint between two distinct variables

    *)
  2. | From_same_struct of {
    1. struct_name : string;
      (*

      The structure type name

      *)
    2. pointer_field_name : string;
      (*

      Name of the constrained pointer field

      *)
    3. size_field_name : string;
      (*

      Name of the constraining size field

      *)
    }
    (*

    Constraint between two fields of the same structure

    *)

A single pointer constraint relates distinct variables or fields in the same structure.

  1. | Pointer_reference
  2. | Pointer_constraint
