Module Sc_report.Types

type config = {
  1. enable : bool;
    (*

    For enabling the report generation

    *)
  2. report_format : [ `Pretty | `OnePage ];
    (*

    The format of the report: pretty report (with tabs and nice css) or one page

    *)
  3. output_dir : string;
    (*

    The directory in which the report is generated

    *)
}

Configuration of the report builder

type config_error =
  1. | Invalid_config_value of {
    1. key : string;
    2. value : string;
    3. possible_values : string list;
    }
exception CONFIG_ERROR of config_error
type resource_error =
  1. | Missing_resource_file of {
    1. filename : string;
    }
exception RESOURCE_ERROR of resource_error
